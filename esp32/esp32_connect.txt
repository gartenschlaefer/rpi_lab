# --
# esp32 connect

# activate idf environment -> my it problems issue
get_idf

# got to folder with project, example project in esp_idf folder
cd ../project_folder

# set target
idf.py set-target esp32

# build
idf.py build

# flash
idf.py flash -p /dev/ttyUSB0

# monitor
idf.py monitor -p /dev/ttyUSB0

# run with emulator qemu: 
# https://docs.espressif.com/projects/esp-idf/en/stable/esp32s3/api-guides/tools/qemu.html

# qemu monitor
idf.py qemu monitor

# if not installled -> install qemu with idf:
python $IDF_PATH/tools/idf_tools.py install qemu-xtensa

# qemu debug
idf.py qemu gdb

# debug and monitor, two terminals
idf.py qemu --gdb monitor
idf.py gdb
